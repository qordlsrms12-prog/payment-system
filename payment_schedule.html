var SHEET_NAME = '지급신청'; function getSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); var headers = ['ID','신청일시','신청자','부서','구분','품목','지급일','구매처','금액','상태','검토자','검토일','검토의견']; sheet.getRange(1, 1, 1, 13).setValues([headers]); sheet.getRange(1, 1, 1, 13).setBackground('#1a3c6e'); sheet.getRange(1, 1, 1, 13).setFontColor('#ffffff'); sheet.getRange(1, 1, 1, 13).setFontWeight('bold'); sheet.setFrozenRows(1); } return sheet; } function toDateStr(val) { if (!val) return ''; if (typeof val === 'string') return val.slice(0, 10); var d = new Date(val); if (isNaN(d.getTime())) return String(val); var m = d.getMonth() + 1; var day = d.getDate(); return d.getFullYear() + '-' + (m < 10 ? '0' + m : '' + m) + '-' + (day < 10 ? '0' + day : '' + day); } function makeOutput(obj, callback) { var json = JSON.stringify(obj); if (callback) { return ContentService.createTextOutput(callback + '(' + json + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON); } function handleGetAll(callback) { var sheet = getSheet(); var lastRow = sheet.getLastRow(); if (lastRow <= 1) { return makeOutput({result: 'success', requests: []}, callback); } var data = sheet.getRange(2, 1, lastRow - 1, 13).getValues(); var list = []; for (var i = 0; i < data.length; i++) { var r = data[i]; if (r[0] === '') continue; list.push({ id: r[0], submittedAt: String(r[1]), requester: r[2], dept: r[3], category: r[4], item: r[5], paydate: toDateStr(r[6]), vendor: r[7], amount: Number(r[8]), status: r[9], reviewedBy: r[10], reviewedAt: r[11] ? toDateStr(r[11]) : '', comment: r[12] }); } return makeOutput({result: 'success', requests: list}, callback); } function handleAdd(p, callback) { var sheet = getSheet(); var lastRow = sheet.getLastRow(); var newId = 1; if (lastRow > 1) { newId = Number(sheet.getRange(lastRow, 1).getValue()) + 1; } var now = new Date(); var hh = now.getHours() < 10 ? '0' + now.getHours() : '' + now.getHours(); var mm = now.getMinutes() < 10 ? '0' + now.getMinutes() : '' + now.getMinutes(); var submittedAt = now.getFullYear() + '-' + (now.getMonth()+1 < 10 ? '0'+(now.getMonth()+1) : ''+(now.getMonth()+1)) + '-' + (now.getDate() < 10 ? '0'+now.getDate() : ''+now.getDate()) + ' ' + hh + ':' + mm; var row = [ newId, submittedAt, p.requester || '', p.dept || '', p.category || '', p.item || '', p.paydate || '', p.vendor || '', Number(p.amount) || 0, '대기중', '', '', '' ]; sheet.appendRow(row); var nr = sheet.getLastRow(); sheet.getRange(nr, 9).setNumberFormat('#,##0'); sheet.getRange(nr, 1, 1, 13).setBackground('#fff3cd'); return makeOutput({result: 'success', id: newId}, callback); } function handleReview(p, callback) { var sheet = getSheet(); var lastRow = sheet.getLastRow(); if (lastRow <= 1) { return makeOutput({result: 'error', message: '데이터 없음'}, callback); } var ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); var targetRow = -1; for (var i = 0; i < ids.length; i++) { if (Number(ids[i][0]) === Number(p.id)) { targetRow = i + 2; break; } } if (targetRow === -1) { return makeOutput({result: 'error', message: '신청을 찾을 수 없습니다'}, callback); } var today = new Date(); var reviewedAt = today.getFullYear() + '-' + (today.getMonth()+1 < 10 ? '0'+(today.getMonth()+1) : ''+(today.getMonth()+1)) + '-' + (today.getDate() < 10 ? '0'+today.getDate() : ''+today.getDate()); sheet.getRange(targetRow, 10).setValue(p.status); sheet.getRange(targetRow, 11).setValue(p.reviewedBy || ''); sheet.getRange(targetRow, 12).setValue(reviewedAt); sheet.getRange(targetRow, 13).setValue(p.comment || ''); var bg = '#fff3cd'; if (p.status === '승인') bg = '#d4edda'; if (p.status === '반려') bg = '#f8d7da'; sheet.getRange(targetRow, 1, 1, 13).setBackground(bg); return makeOutput({result: 'success'}, callback); } function handleClearAll(callback) { var sheet = getSheet(); var lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.deleteRows(2, lastRow - 1); } return makeOutput({result: 'success'}, callback); } function doGet(e) { var p = e.parameter; var callback = p.callback || ''; var action = p.action || ''; if (action === 'getAll') return handleGetAll(callback); if (action === 'add') return handleAdd(p, callback); if (action === 'review') return handleReview(p, callback); if (action === 'clearAll') return handleClearAll(callback); return makeOutput({result: 'error', message: 'unknown action'}, callback); } function doPost(e) { var p = e.parameter || {}; var callback = p.callback || ''; var action = p.action || ''; if (action === 'getAll') return handleGetAll(callback); if (action === 'add') return handleAdd(p, callback); if (action === 'review') return handleReview(p, callback); if (action === 'clearAll') return handleClearAll(callback); return makeOutput({result: 'error', message: 'unknown action'}, callback); }
